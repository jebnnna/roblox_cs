-- UTIL
local function GetHRP()
    Character = LocalPlayer.Character
    HRP = Character and Character:FindFirstChild("HumanoidRootPart")
    return HRP
end

local function DistanceTo(inst)
    local root = GetHRP(); if not root then return 0 end
    local pos
    if inst:IsA("BasePart") then pos = inst.Position
    elseif inst:IsA("Model") then
        local p = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
        if p then pos = p.Position end
    end
    return pos and math.floor((root.Position - pos).Magnitude) or 0
end

-- RunSearch
local function PassesFilter(inst)
    if S.FilterMode=="All" then return inst:IsA("BasePart") or inst:IsA("Model")
    elseif S.FilterMode=="Parts" then return inst:IsA("BasePart")
    elseif S.FilterMode=="Models" then return inst:IsA("Model") end
end

local function RunSearch(query)
    ClearHL(); lastResults = {}
    if query=="" then StatusLbl.Text="  type a name first"; BuildResultRows(); return end
    local ql = query:lower()
    hlFolder = Instance.new("Folder"); hlFolder.Name="Scanner_HL"; hlFolder.Parent=workspace
    local seen = {}
    for _, inst in ipairs(workspace:GetDescendants()) do
        if seen[inst] or not inst.Name:lower():find(ql,1,true) or not PassesFilter(inst) then continue end
        seen[inst] = true
        local hl = Instance.new("Highlight")
        hl.FillColor=Color3.fromRGB(200,100,255); hl.OutlineColor=Color3.fromRGB(255,255,255)
        hl.FillTransparency=0.5; hl.OutlineTransparency=0
        hl.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop; hl.Adornee=inst; hl.Parent=hlFolder
        local lb2 = inst:IsA("BasePart") and inst or (inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart"))
        if lb2 then
            local bb=Instance.new("BillboardGui"); bb.Size=UDim2.new(0,120,0,24)
            bb.StudsOffset=Vector3.new(0,4,0); bb.AlwaysOnTop=true; bb.Adornee=lb2; bb.Parent=hlFolder
            local tl=Instance.new("TextLabel",bb); tl.Text=inst.Name; tl.Size=UDim2.new(1,0,1,0)
            tl.BackgroundTransparency=1; tl.TextColor3=Color3.fromRGB(200,100,255)
            tl.TextStrokeTransparency=0; tl.TextStrokeColor3=Color3.fromRGB(0,0,0)
            tl.Font=Enum.Font.GothamBold; tl.TextSize=12
        end
        lastResults[#lastResults+1]={instance=inst,name=inst.Name,class=inst.ClassName,dist=DistanceTo(inst)}
    end
    table.sort(lastResults,function(a,b) return a.dist<b.dist end)
    StatusLbl.Text="  "..#lastResults.." ("..S.FilterMode..") for: "..query
    BuildResultRows()
end

local function RunRadarScan()
    local root=GetHRP(); if not root then return end
    ClearRadarHL()
    radarFolder=Instance.new("Folder"); radarFolder.Name="Scanner_Radar"; radarFolder.Parent=workspace
    local myPos=root.Position; local count=0; local seen={}
    for _,inst in ipairs(workspace:GetDescendants()) do
        if seen[inst] then continue end
        if Character and inst:IsDescendantOf(Character) then continue end
        local pos
        if inst:IsA("BasePart") then pos=inst.Position
        elseif inst:IsA("Model") then
            local p=inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart"); if p then pos=p.Position end
        end
        if not pos or (myPos-pos).Magnitude>S.RadarRadius then continue end
        seen[inst]=true
        local hl=Instance.new("Highlight")
        hl.FillColor=Color3.fromRGB(80,200,255); hl.OutlineColor=Color3.fromRGB(255,255,255)
        hl.FillTransparency=0.55; hl.OutlineTransparency=0
        hl.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop; hl.Adornee=inst; hl.Parent=radarFolder
        local lb2=inst:IsA("BasePart") and inst or (inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart"))
        if lb2 then
            local bb=Instance.new("BillboardGui"); bb.Size=UDim2.new(0,140,0,24)
            bb.StudsOffset=Vector3.new(0,4,0); bb.AlwaysOnTop=true; bb.Adornee=lb2; bb.Parent=radarFolder
            local tl=Instance.new("TextLabel",bb)
            tl.Text=inst.Name.." ["..math.floor((myPos-pos).Magnitude).."st]"
            tl.Size=UDim2.new(1,0,1,0); tl.BackgroundTransparency=1
            tl.TextColor3=Color3.fromRGB(80,200,255); tl.TextStrokeTransparency=0
            tl.TextStrokeColor3=Color3.fromRGB(0,0,0); tl.Font=Enum.Font.GothamBold; tl.TextSize=11
        end
        count=count+1
    end
    RadarStatusLbl.Text="  "..count.." within "..S.RadarRadius.."st"
end

local function StartRadar()
    S.RadarActive=true; RadarLbl.Text="Radar: ON"; RadarLbl.TextColor3=Color3.fromRGB(80,200,255)
    RadarBtn.BackgroundColor3=Color3.fromRGB(35,35,35)
    if radarConn then radarConn:Disconnect(); radarConn=nil end
    RunRadarScan()
    radarConn=RunService.Heartbeat:Connect(function(dt)
        radarTick=radarTick+dt; if radarTick>=0.5 then radarTick=0; RunRadarScan() end
    end)
end

local function StopRadar()
    S.RadarActive=false
    if radarConn then radarConn:Disconnect(); radarConn=nil end
    ClearRadarHL(); RadarLbl.Text="Radar: OFF"; RadarLbl.TextColor3=Color3.fromRGB(210,210,210)
    RadarBtn.BackgroundColor3=Color3.fromRGB(25,20,40); RadarStatusLbl.Text="  radar off"
end

RadarBtn.MouseButton1Click:Connect(function()
    if S.RadarActive then StopRadar() else StartRadar() end
end)

-- 9. Continuous Teleport
local ContBtn, ContLbl = MkBtn("Continuous Teleport: OFF", Color3.fromRGB(20,30,45), Color3.fromRGB(40,80,120), Color3.fromRGB(100,180,255))
ContBtn.MouseButton1Click:Connect(function()
    if contRunning then
        contRunning=false; ContLbl.Text="Continuous Teleport: OFF"; ContLbl.TextColor3=Color3.fromRGB(100,180,255)
        StatusLbl.Text="  stopped"; return
    end
    if #lastResults==0 then StatusLbl.Text="  search first to build the list"; return end
    contRunning=true; ContLbl.Text="Continuous Teleport: ON  ◼"; ContLbl.TextColor3=Color3.fromRGB(255,100,100)
    task.spawn(function()
        while contRunning do
            -- Build pool from current list — favs only, or all if none starred
            local pool={}
            for _,e in ipairs(lastResults) do if favorites[e.instance] then pool[#pool+1]=e end end
            if #pool==0 then pool=lastResults end
            for i,entry in ipairs(pool) do
                if not contRunning then break end
                local root=GetHRP(); if not root then contRunning=false; break end
                local inst=entry.instance
                if inst and inst.Parent then
                    local pos
                    if inst:IsA("BasePart") then pos=inst.Position
                    elseif inst:IsA("Model") then
                        local p=inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
                        if p then pos=p.Position end
                    end
                    if pos then
                        root.CFrame=CFrame.new(pos)*CFrame.new(0,4,0)
                        StatusLbl.Text="  ["..i.."/"..#pool.."] "..entry.name
                    end
                end
                task.wait(S.TpDelay)
            end
            -- Loop restarts from top of pool — no re-search, same list
        end
        ContLbl.Text="Continuous Teleport: OFF"; ContLbl.TextColor3=Color3.fromRGB(100,180,255)
    end)
end)

        elseif inst:IsA("Model") then

        existing[inst]=true
        lastResults[#lastResults+1]={instance=inst,name=inst.Name,class=inst.ClassName,dist=math.floor((myPos-pos).Magnitude)}
        added=added+1
    end
    table.sort(lastResults,function(a,b) return a.dist<b.dist end)
    BuildResultRows(); StatusLbl.Text="  added "..added.." ("..#lastResults.." total)"
end)

-- DRAG — on MainFrame level so it works even when content is scrolling
local drag, dStart, dPos = false, nil, nil
MainFrame.InputBegan:Connect(function(i)
    if (i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1)
    and i.Position.Y - MainFrame.AbsolutePosition.Y < 44 then
        drag=true; dStart=i.Position; dPos=MainFrame.Position
        i.Changed:Connect(function() if i.UserInputState==Enum.UserInputState.End then drag=false end end)
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if drag and (i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseMovement) then
        local d=i.Position-dStart
        MainFrame.Position=UDim2.new(dPos.X.Scale,dPos.X.Offset+d.X,dPos.Y.Scale,dPos.Y.Offset+d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then
        drag=false
    end
end)
