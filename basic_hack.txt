-- FLY
local flyBV, flyBG

local function StopFly()
    if Connections.Fly then Connections.Fly:Disconnect() Connections.Fly = nil end
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
    if Humanoid then Humanoid:ChangeState(Enum.HumanoidStateType.Freefall) end
end

local function StartFly()
    StopFly()
    Character        = LocalPlayer.Character
    HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
    if not HumanoidRootPart then return end

    flyBV = Instance.new("BodyVelocity")
    flyBV.Velocity  = Vector3.zero
    flyBV.MaxForce  = Vector3.new(1e5, 1e5, 1e5)
    flyBV.Parent    = HumanoidRootPart

    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
    flyBG.P         = 1e4
    flyBG.Parent    = HumanoidRootPart

    Connections.Fly = RunService.Heartbeat:Connect(function()
        if not (Character and HumanoidRootPart) then return end
        local cam = workspace.CurrentCamera

        if State.TargetPlayer and State.TargetPlayerName ~= "" then
            local targetPlr = Players:FindFirstChild(State.TargetPlayerName)
            if targetPlr and targetPlr.Character then
                local tRoot = targetPlr.Character:FindFirstChild("HumanoidRootPart")
                if tRoot then
                    local dir  = (tRoot.Position - HumanoidRootPart.Position)
                    local dist = dir.Magnitude
                    if dist > 3 then
                        flyBV.Velocity = dir.Unit * State.FlySpeed
                        flyBG.CFrame   = CFrame.new(HumanoidRootPart.Position, tRoot.Position)
                    else
                        flyBV.Velocity = Vector3.zero
                    end
                    return
                end
            end
        end

        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W)         then dir = dir + cam.CFrame.LookVector  end
        if UserInputService:IsKeyDown(Enum.KeyCode.S)         then dir = dir - cam.CFrame.LookVector  end
        if UserInputService:IsKeyDown(Enum.KeyCode.A)         then dir = dir - cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D)         then dir = dir + cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space)     then dir = dir + Vector3.new(0, 1, 0)  end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0)  end
        flyBV.Velocity = dir.Magnitude > 0 and dir.Unit * State.FlySpeed or Vector3.zero
        flyBG.CFrame   = cam.CFrame
    end)
end

-- ─── LAUNCH ON TOUCH — NDS REWRITE ───────────────────────────
--
-- WHY THE ORIGINAL FAILED:
--   BodyVelocity on another player's HumanoidRootPart is local-only under
--   FilteringEnabled. The target never actually flies on their screen.
--
-- HOW THIS VERSION WORKS (three independent layers):
--
--   Layer A — BodyVelocity on target (executor-level bonus)
--     Wrapped in pcall. Local effect is guaranteed. Server-side replication
--     depends on Delta's privilege level — treat it as a bonus, not the primary.
--
--   Layer B — PlatformStand on target
--     Disables their Humanoid input for ~0.9s. Compounds Layer A if it lands.
--     Also pcall-guarded.
--
--   Layer C — Self-propulsion dash (THE reliable layer)
--     YOUR character physics is network-owned by you, so it ALWAYS replicates.
--     NDS is a 2008 game that predates Roblox's collision group system — characters
--     physically collide with each other (players can stand on each other, get bumped
--     during disasters). A high-velocity dash into the target produces a real
--     server-side collision impulse that sends them flying even if Layers A and B
--     are FE-blocked.
--     The dash lasts ~0.055s (just long enough to cover 5 studs at 260 studs/s),
--     then the HRP snaps back to the saved CFrame so you barely move on screen.
--
--   NoClip interaction:
--     If NoClip is ON, your character has CanCollide=false, so the dash produces
--     no physical collision. Layer C is skipped automatically; Layers A and B
--     remain as the only attempt.
--
-- DETECTION — two signals, both active simultaneously:
--   Primary:  Touched events on every local character limb (fires on real physics contact)
--   Fallback: Heartbeat proximity scan at 5-stud range (catches lag/missed Touched events)
--   Per-player cooldown: 0.45s so no target gets spammed every frame
--
-- LIFETIME MANAGEMENT:
--   Touched connections stored in touchConns[], rebuilt on respawn via CharacterAdded.
--   DisconnectAll() and ResetAllStates() both clean up touchConns.

local launchCooldowns = {}
local touchConns      = {}  -- per-limb Touched connections, rebuilt each respawn

-- LaunchTarget: fires all three layers against one target
local function LaunchTarget(targetRoot, myHRP)
    if not (targetRoot and myHRP and targetRoot.Parent and myHRP.Parent) then return end

    local delta   = targetRoot.Position - myHRP.Position
    local pushDir = delta.Magnitude > 0.1 and delta.Unit
                    or Vector3.new(math.random(-1, 1), 0.3, math.random(-1, 1))

    -- Layer A: BodyVelocity on target root
    pcall(function()
        targetRoot.Anchored = false
        local bv = Instance.new("BodyVelocity")
        bv.Velocity = pushDir * 220 + Vector3.new(0, 130, 0)
        bv.MaxForce = Vector3.new(1e7, 1e7, 1e7)
        bv.P        = 1e6
        bv.Parent   = targetRoot
        Debris:AddItem(bv, 0.18)
    end)

    -- Layer B: PlatformStand — kills their input briefly
    pcall(function()
        local hum = targetRoot.Parent:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.PlatformStand = true
            task.delay(0.9, function()
                pcall(function() hum.PlatformStand = false end)
            end)
        end
    end)

    -- Layer C: Self-propulsion dash
    -- Skipped when NoClip is ON (CanCollide=false means no physical hit)
    if not State.NoClip and myHRP and myHRP.Parent then
        local savedCF = myHRP.CFrame

        local dash = Instance.new("BodyVelocity")
        dash.Velocity = pushDir * 260 + Vector3.new(0, 15, 0)
        dash.MaxForce = Vector3.new(1e6, 1e6, 1e6)
        dash.P        = 1e5
        dash.Parent   = myHRP
        Debris:AddItem(dash, 0.055)   -- auto-remove after ~3 physics ticks

        -- Snap back so we don't go flying with them
        task.delay(0.07, function()
            pcall(function()
                if myHRP and myHRP.Parent then
                    myHRP.CFrame = savedCF
                end
            end)
        end)
    end
end

-- SetupTouched: attach Touched to every limb of the current character
local function SetupTouched()
    for _, c in pairs(touchConns) do pcall(function() c:Disconnect() end) end
    touchConns = {}
    local char = LocalPlayer.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            local conn = part.Touched:Connect(function(hit)
                if not State.TouchLaunch then return end
                local targetPlr = Players:GetPlayerFromCharacter(hit.Parent)
                if not targetPlr or targetPlr == LocalPlayer then return end
                local root = hit.Parent:FindFirstChild("HumanoidRootPart")
                if not root then return end
                local hum = hit.Parent:FindFirstChildOfClass("Humanoid")
                if not hum or hum.Health <= 0 then return end
                local now = tick()
                if (now - (launchCooldowns[targetPlr.UserId] or 0)) < 0.45 then return end
                launchCooldowns[targetPlr.UserId] = now
                local myHRP = LocalPlayer.Character
                    and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if myHRP then LaunchTarget(root, myHRP) end
            end)
            table.insert(touchConns, conn)
        end
    end
end

CreateToggle("Launch on Touch", "TouchLaunch", function(on)
    if Connections.TouchLaunch then Connections.TouchLaunch:Disconnect() Connections.TouchLaunch = nil end
    for _, c in pairs(touchConns) do pcall(function() c:Disconnect() end) end
    touchConns      = {}
    launchCooldowns = {}

    if on then
        SetupTouched()  -- primary detection: Touched events on all limbs

        local lastRebuild = tick()
        Connections.TouchLaunch = RunService.Heartbeat:Connect(function()
            Character        = LocalPlayer.Character
            HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
            if not (Character and HumanoidRootPart) then return end

            -- Rebuild Touched hooks if character reference drifted (edge-case respawn)
            local now = tick()
            if now - lastRebuild > 2 then
                lastRebuild = now
                SetupTouched()
            end

            -- Heartbeat fallback: proximity scan at 5 studs
            -- Catches targets within range even if Touched misfires under lag
            local myPos = HumanoidRootPart.Position
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr == LocalPlayer or not plr.Character then continue end
                local hum  = plr.Character:FindFirstChildOfClass("Humanoid")
                if not hum or hum.Health <= 0 then continue end
                local root = plr.Character:FindFirstChild("HumanoidRootPart")
                if not root then continue end
                local dist    = (root.Position - myPos).Magnitude
                local lastHit = launchCooldowns[plr.UserId] or 0
                if dist < 5 and (now - lastHit) > 0.45 then
                    launchCooldowns[plr.UserId] = now
                    LaunchTarget(root, HumanoidRootPart)
                end
            end
        end)
    end
end)

        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end

    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

-- ─── DEATH RESET ──────────────────────────────────────────────
local function ResetAllStates()
    DisconnectAll()
    StopFly()
    ClearESP()
    launchCooldowns = {}
    for _, c in pairs(touchConns) do pcall(function() c:Disconnect() end) end
    touchConns = {}
end

-- ─── CHARACTER ADDED ──────────────────────────────────────────
LocalPlayer.CharacterAdded:Connect(function(char)
    Character        = char
    Humanoid         = char:WaitForChild("Humanoid")
    HumanoidRootPart = char:WaitForChild("HumanoidRootPart")

    local saved = {
        Speed        = State.Speed,
        InfJump      = State.InfJump,
        JumpTp       = State.JumpTp,
        Fly          = State.Fly,
        NoClip       = State.NoClip,
        ESP          = State.ESP,
        AntiAFK      = State.AntiAFK,
        TouchLaunch  = State.TouchLaunch,
        TargetPlayer = State.TargetPlayer,
    }

    ResetAllStates()

    task.delay(0.6, function()
        for key, wasOn in pairs(saved) do
            if wasOn then
                State[key] = true
                local reg  = ToggleRegistry[key]
                if reg then
                    reg.UpdateVisual()
                    if reg.Callback then reg.Callback(true) end
                end
            end
        end
        -- Rebuild Touched hooks after respawn if TouchLaunch was active
        -- (CharacterAdded fires before the callback re-runs, so SetupTouched
        --  would have run on the old character — rebuild it on the new one)
        if State.TouchLaunch then SetupTouched() end
    end)
end)
