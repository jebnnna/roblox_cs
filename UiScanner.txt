-- ─── CLICK ENGINE ────────────────────────────────────────────
-- Four-method chain: TextButton, ImageButton, any GuiObject,
-- hidden menus (ancestors invisible), off-screen elements.
-- Method 1: firesignal (executor API) — visibility/position irrelevant
-- Method 2: firebutton (alternate executor API)
-- Method 3: VIM tap with full ancestor chain revealed
local VIM = game:GetService("VirtualInputManager")

local function FireElement(inst)
    if not inst or not inst.Parent then return false end

    if inst:IsA("GuiButton") then
        -- Method 1: firesignal — Delta executor API, works on TextButton AND
        -- ImageButton, hidden or visible, on-screen or off. Fires Down/Up/Click
        -- AND Activated because some games wire only Activated internally.
        local ok = false
        pcall(function()
            firesignal(inst.MouseButton1Down, 0, 0)
            firesignal(inst.MouseButton1Up,   0, 0)
            firesignal(inst.MouseButton1Click)
            firesignal(inst.Activated)
            ok = true
        end)
        if ok then return true end

        -- Method 2: firebutton — alternate executor API, same effect
        pcall(function()
            if type(firebutton) == "function" then
                firebutton(inst)
                ok = true
            end
        end)
        if ok then return true end
    end

    -- Method 3: VIM tap with full ancestry chain revealed.
    -- Problem with the old code: it only showed the element itself, so if any
    -- parent Frame was hidden the AbsoluteSize was still (0,0) and the tap
    -- fired at wrong coordinates. This version walks all ancestors, reveals
    -- every hidden GuiObject up the chain, waits one frame so Roblox
    -- recalculates AbsolutePosition/Size, taps, then restores everything.
    if inst:IsA("GuiObject") then
        pcall(function()
            local hidden = {}
            local p = inst.Parent
            while p and p ~= game do
                if p:IsA("GuiObject") and not p.Visible then
                    hidden[#hidden+1] = p
                    p.Visible = true
                end
                p = p.Parent
            end
            local wasVisible = inst.Visible
            inst.Visible = true

            task.wait()  -- one frame: AbsolutePosition/Size recalculates

            local ap = inst.AbsolutePosition
            local as = inst.AbsoluteSize
            if as.X > 0 and as.Y > 0 then
                local cx = ap.X + as.X * 0.5
                local cy = ap.Y + as.Y * 0.5
                VIM:SendMouseButtonEvent(cx, cy, 0, true,  game, 0)
                task.wait(0.05)
                VIM:SendMouseButtonEvent(cx, cy, 0, false, game, 0)
            end

            inst.Visible = wasVisible
            for _, obj in ipairs(hidden) do
                pcall(function() obj.Visible = false end)
            end
        end)
    end

    return true
end

-- ─── HIGHLIGHT ENGINE ────────────────────────────────────────
local function ClearHL()
    for _, obj in ipairs(hlInstances) do pcall(function() obj:Destroy() end) end
    hlInstances = {}
end

local function HighlightUI(inst)
    pcall(function()
        if inst:FindFirstChild("__UHL") then inst.__UHL:Destroy() end
        local sk = Instance.new("UIStroke", inst)
        sk.Name = "__UHL"; sk.Color = Color3.fromRGB(255,100,200); sk.Thickness = 3
        hlInstances[#hlInstances+1] = sk
    end)
end

-- ─── SEARCH ──────────────────────────────────────────────────
local FilterModes  = {"All","Buttons","TextLabels","Frames","ImageButtons"}
local FilterColors = {
    All=Color3.fromRGB(210,210,210), Buttons=Color3.fromRGB(255,170,80),
    TextLabels=Color3.fromRGB(100,200,255), Frames=Color3.fromRGB(180,255,100),
    ImageButtons=Color3.fromRGB(255,100,200),
}

local function PassesFilter(inst)
    local m = S.FilterMode
    if m=="All"          then return inst:IsA("GuiObject")
    elseif m=="Buttons"  then return inst:IsA("TextButton") or inst:IsA("ImageButton")
    elseif m=="TextLabels" then return inst:IsA("TextLabel")
    elseif m=="Frames"   then return inst:IsA("Frame") or inst:IsA("ScrollingFrame")
    elseif m=="ImageButtons" then return inst:IsA("ImageButton") end
    return false
end

local function AddToList(inst)
    for _, e in ipairs(lastResults) do if e.instance==inst then return end end
    local txt = ""
    if inst:IsA("TextLabel") or inst:IsA("TextButton") then txt = inst.Text:sub(1,28) end
    lastResults[#lastResults+1] = {instance=inst, name=inst.Name, class=inst.ClassName, text=txt}
    HighlightUI(inst)
    BuildResultRows()
end

local function RunSearch(query)
    ClearHL(); lastResults = {}
    if query=="" then StatusLbl.Text="  type a name first"; BuildResultRows(); return end
    local ql = query:lower()
    local roots = {LocalPlayer:WaitForChild("PlayerGui")}
    if S.ScanCoreGui then roots[#roots+1] = game:GetService("CoreGui") end
    for _, root in ipairs(roots) do
        for _, inst in ipairs(root:GetDescendants()) do
            if inst:IsDescendantOf(ScreenGui) then continue end
            if not inst.Name:lower():find(ql,1,true) then continue end
            if not PassesFilter(inst) then continue end
            -- Skip invisible unless IncludeHidden is ON
            if not S.IncludeHidden then
                -- Walk up ancestry — if any ancestor is invisible, skip
                local visible = true
                local p = inst
                while p and p ~= root do
                    if p:IsA("GuiObject") and not p.Visible then
                        visible = false; break
                    end
                    p = p.Parent
                end
                if not visible then continue end
            end
            local txt=""
            if inst:IsA("TextLabel") or inst:IsA("TextButton") then txt=inst.Text:sub(1,28) end
            lastResults[#lastResults+1]={instance=inst,name=inst.Name,class=inst.ClassName,text=txt}
            HighlightUI(inst)
        end
    end
    StatusLbl.Text="  "..#lastResults.." result"..(#lastResults==1 and "" or "s").." ("..S.FilterMode..") for: "..query
    BuildResultRows()
end

-- ─── HOVER DETECTION ─────────────────────────────────────────
-- Uses UserInputService to track cursor/touch position every frame,
-- finds which GuiButton is under it and adds it to the list
local HoverBtn, HoverLbl = Btn("Hover Detection: OFF", Color3.fromRGB(18,20,30), Color3.fromRGB(50,50,80), Color3.fromRGB(150,150,220))
local hoverActive = false

HoverBtn.MouseButton1Click:Connect(function()
    hoverActive = not hoverActive
    HoverLbl.Text = "Hover Detection: "..(hoverActive and "ON" or "OFF")
    HoverLbl.TextColor3 = hoverActive and Color3.fromRGB(150,200,255) or Color3.fromRGB(150,150,220)
    if hoverActive then
        if hoverConn then hoverConn:Disconnect() end
        hoverConn = RunService.Heartbeat:Connect(function()
            local pos = UserInputService:GetMouseLocation()
            local pg = LocalPlayer:FindFirstChild("PlayerGui")
            if not pg then return end
            -- GetGuiObjectsAtPosition returns all GuiObjects under the cursor
            local objs = pg:GetGuiObjectsAtPosition(pos.X, pos.Y)
            for _, obj in ipairs(objs) do
                if obj:IsDescendantOf(ScreenGui) then continue end
                if obj:IsA("GuiButton") then
                    AddToList(obj)
                end
            end
        end)
    else
        if hoverConn then hoverConn:Disconnect(); hoverConn=nil end
        StatusLbl.Text = "  hover detection off"
    end
end)

-- ─── CLICK DETECTION ─────────────────────────────────────────
-- Connects MouseButton1Click to every GuiButton in PlayerGui.
-- When any fires, that button is added to the list.
local ClickDetBtn, ClickDetLbl = Btn("Click Detection: OFF", Color3.fromRGB(18,20,30), Color3.fromRGB(50,60,50), Color3.fromRGB(150,220,150))
local clickDetActive = false

local function StopClickDetection()
    clickDetActive = false
    for _, c in ipairs(clickDetConns) do pcall(function() c:Disconnect() end) end
    clickDetConns = {}
    ClickDetLbl.Text = "Click Detection: OFF"
    ClickDetLbl.TextColor3 = Color3.fromRGB(150,220,150)
end

local function StartClickDetection()
    StopClickDetection()
    clickDetActive = true
    ClickDetLbl.Text = "Click Detection: ON  ◼ stop"
    ClickDetLbl.TextColor3 = Color3.fromRGB(100,255,100)
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then StatusLbl.Text="  no PlayerGui"; return end
    local function WireButton(btn)
        if btn:IsDescendantOf(ScreenGui) then return end
        local c = btn.MouseButton1Click:Connect(function()
            AddToList(btn)
            StatusLbl.Text = "  detected click: "..btn.Name
        end)
        clickDetConns[#clickDetConns+1] = c
    end
    for _, inst in ipairs(pg:GetDescendants()) do
        if inst:IsA("GuiButton") then WireButton(inst) end
    end
    -- Also wire new buttons that appear after detection starts
    local addedConn = pg.DescendantAdded:Connect(function(inst)
        if inst:IsA("GuiButton") then
            task.defer(function() if clickDetActive then WireButton(inst) end end)
        end
    end)
    clickDetConns[#clickDetConns+1] = addedConn
    StatusLbl.Text = "  listening for clicks in PlayerGui..."
end

ClickDetBtn.MouseButton1Click:Connect(function()
    if clickDetActive then StopClickDetection() else StartClickDetection() end
end)

local AutoBtn, AutoLbl = Btn("Continuous Click: OFF", Color3.fromRGB(20,30,45), Color3.fromRGB(40,80,120), Color3.fromRGB(100,180,255))
AutoBtn.MouseButton1Click:Connect(function()
    if autoRunning then
        autoRunning=false; AutoLbl.Text="Continuous Click: OFF"; AutoLbl.TextColor3=Color3.fromRGB(100,180,255)
        StatusLbl.Text="  auto-click stopped"; return
    end
    local query=SearchBox.Text
    if query=="" and #lastResults==0 then StatusLbl.Text="  search or detect first"; return end
    autoRunning=true; AutoLbl.Text="Continuous Click: ON  ◼ stop"; AutoLbl.TextColor3=Color3.fromRGB(255,100,100)
    task.spawn(function()
        while autoRunning do
            if query~="" then RunSearch(query); BuildResultRows() end
            local pool={}
            for _,e in ipairs(lastResults) do if favorites[e.instance] then pool[#pool+1]=e end end
            if #pool==0 then pool=lastResults end
            if #pool==0 then StatusLbl.Text="  no results, retrying..."; task.wait(S.ClickDelay)
            else
                for i,entry in ipairs(pool) do
                    if not autoRunning then break end
                    if entry.instance and entry.instance.Parent then
                        FireElement(entry.instance)
                        StatusLbl.Text="  ["..i.."/"..#pool.."] "..entry.name
                    end
                    task.wait(S.ClickDelay)
                end
            end
        end
        AutoLbl.Text="Continuous Click: OFF"; AutoLbl.TextColor3=Color3.fromRGB(100,180,255)
    end)
end)

CountBtn2.MouseButton1Click:Connect(function()
    local btns,lbls,frames,imgs,tot=0,0,0,0,0
    local pg=LocalPlayer:FindFirstChild("PlayerGui"); if not pg then StatusLbl.Text="  no PlayerGui" return end
    for _,i in ipairs(pg:GetDescendants()) do
        tot=tot+1
        if i:IsA("TextButton") or i:IsA("ImageButton") then btns=btns+1
        elseif i:IsA("TextLabel") then lbls=lbls+1
        elseif i:IsA("Frame") then frames=frames+1
        elseif i:IsA("ImageLabel") then imgs=imgs+1 end
    end
    StatusLbl.Text=string.format("  Btns:%d  Lbls:%d  Frames:%d  Imgs:%d  Total:%d",btns,lbls,frames,imgs,tot)
end)

-- ─── DRAG ────────────────────────────────────────────────────
local drag, dStart, dPos = false, nil, nil
MainFrame.InputBegan:Connect(function(i)
    if (i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1)
    and i.Position.Y - MainFrame.AbsolutePosition.Y < 44 then
        drag=true; dStart=i.Position; dPos=MainFrame.Position
        i.Changed:Connect(function() if i.UserInputState==Enum.UserInputState.End then drag=false end end)
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if drag and (i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseMovement) then
        local d=i.Position-dStart
        MainFrame.Position=UDim2.new(dPos.X.Scale,dPos.X.Offset+d.X,dPos.Y.Scale,dPos.Y.Offset+d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType==Enum.UserInputType.Touch or i.UserInputType==Enum.UserInputType.MouseButton1 then
        drag=false
    end
end)

-- ─── DEATH PERSISTENCE ───────────────────────────────────────
LocalPlayer.CharacterAdded:Connect(function()
    local wasAuto = autoRunning
    autoRunning = false
    task.delay(1, function()
        if wasAuto and not autoRunning then
            -- Re-start auto click loop directly
            local query = SearchBox.Text
            autoRunning = true
            AutoLbl.Text = "Continuous Click: ON  ◼ stop"
            AutoLbl.TextColor3 = Color3.fromRGB(255,100,100)
            task.spawn(function()
                while autoRunning do
                    if query ~= "" then RunSearch(query) end
                    local pool = {}
                    for _, e in ipairs(lastResults) do
                        if favorites[e.instance] then pool[#pool+1] = e end
                    end
                    if #pool == 0 then pool = lastResults end
                    if #pool == 0 then
                        StatusLbl.Text = "  no results, retrying..."
                        task.wait(S.ClickDelay)
                    else
                        for i, entry in ipairs(pool) do
                            if not autoRunning then break end
                            if entry.instance and entry.instance.Parent then
                                FireElement(entry.instance)
                                StatusLbl.Text = "  [" .. i .. "/" .. #pool .. "] " .. entry.name
                            end
                            task.wait(S.ClickDelay)
                        end
                    end
                end
                AutoLbl.Text = "Continuous Click: OFF"
                AutoLbl.TextColor3 = Color3.fromRGB(100,180,255)
            end)
        end
    end)
end)
