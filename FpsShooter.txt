-- ─── STANDARD AIMBOT ─────────────────────────────────────────────────────────
local function GetClosestTarget()
    local vCenter  = Camera.ViewportSize / 2
    local bestDist = State.AimbotFOV
    local bestBone = nil
    local boneName = "Head" -- AA always locks head; regular aimbot respects AimHead flag below
    if not State.AbsoluteAim then
        boneName = State.AimHead and "Head" or "HumanoidRootPart"
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not plr.Character then continue end
        if State.TeamCheck and plr.Team == LocalPlayer.Team then continue end
        local char = plr.Character
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local bone = char:FindFirstChild(boneName)
        if hum and hum.Health > 0 and bone then
            local screenPos, onScreen = Camera:WorldToViewportPoint(bone.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - vCenter).Magnitude
                if dist < bestDist then
                    bestDist = dist
                    bestBone = bone
                end
            end
        end
    end
    return bestBone
end

RunService.RenderStepped:Connect(function()
    -- standard aimbot only runs when AbsoluteAim is off
    if not State.Aimbot or State.AbsoluteAim then return end
    local target = GetClosestTarget()
    if target then
        local goal = CFrame.new(Camera.CFrame.Position, target.Position)
        Camera.CFrame = Camera.CFrame:Lerp(goal, State.AimbotSmooth)
    end
end)

-- ─── ABSOLUTE AIM ENGINE ─────────────────────────────────────────────────────
--
-- How it works:
--   1. RebuildTargetList() scans ALL players, living, any position (not FOV-gated).
--      Each entry is {player, head_bone}. List is sorted by 3D distance so
--      nearest goes first, but AA_CycleTargets makes it cycle through the rest
--      sequentially rather than always snapping back to nearest.
--
--   2. RenderStepped drives the camera — always toward the current target's Head.
--      Smooth = AA_LockSmooth (snappier than standard aimbot so it actually locks
--      in first-person games where camera tracks instantly).
--
--   3. When AA_CycleTargets is on, after the current target's Humanoid.Health
--      reaches 0 (or they leave), AdvanceTarget() increments the index.
--      This means if target A dies the engine slides to target B without you
--      touching anything.
--
--   4. AA_AutoAttack fires every AA_AttackDelay seconds on Heartbeat.
--      It uses VirtualUser:CaptureController() + Button1Down at the center of
--      the screen — that triggers whatever tool the player is holding in first-
--      person. Works for most FPS games on Roblox that use default mouse-click
--      to fire. Games with custom input hooks may need firesignal instead
--      (see note at bottom of this section).
--
--   5. AA_ShowESP builds ESP + Highlight when the engine starts so dj can see
--      every enemy through walls while AA handles the rest.

local function RebuildTargetList()
    aaTargetList = {}
    local hrp = HumanoidRootPart
    if not hrp then return end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not plr.Character then continue end
        if State.TeamCheck and plr.Team == LocalPlayer.Team then continue end
        local char = plr.Character
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local head = char:FindFirstChild("Head")
        local root = char:FindFirstChild("HumanoidRootPart")
        -- include even targets NOT on screen — AbsoluteAim sees through walls
        if hum and hum.Health > 0 and head and root then
            local dist = (root.Position - hrp.Position).Magnitude
            table.insert(aaTargetList, { player = plr, bone = head, dist = dist })
        end
    end
    -- sort nearest first
    table.sort(aaTargetList, function(a, b) return a.dist < b.dist end)
    aaTargetIndex = 1
end

local function GetCurrentAATarget()
    if #aaTargetList == 0 then return nil end
    -- clamp index
    if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
    local entry = aaTargetList[aaTargetIndex]
    -- validate still alive
    local hum = entry.player.Character
        and entry.player.Character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        -- dead — advance and return next
        aaTargetIndex = aaTargetIndex + 1
        if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
        return GetCurrentAATarget()
    end
    return entry.bone
end

local function AdvanceTarget()
    if #aaTargetList == 0 then return end
    aaTargetIndex = aaTargetIndex + 1
    if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
end

local function StartAbsoluteAim()
    -- stop if already running
    if aaConn then aaConn:Disconnect() aaConn = nil end
    if aaAttackConn then aaAttackConn:Disconnect() aaAttackConn = nil end

    -- force visuals on if setting says so
    if State.AA_ShowESP then
        BuildESP()
        BuildHL()
        -- mirror State so the toggle buttons show ON
        State.ESP     = true
        State.ESPGlow = true
        if ToggleRegistry["ESP"]     then ToggleRegistry["ESP"].update()     end
        if ToggleRegistry["ESPGlow"] then ToggleRegistry["ESPGlow"].update() end
    end

    -- initial scan
    RebuildTargetList()

    -- rebuild list every 3s in case players join/die
    task.spawn(function()
        while State.AbsoluteAim do
            task.wait(3)
            if State.AbsoluteAim then
                RebuildTargetList()
            end
        end
    end)

    -- RenderStepped: camera lock to current target Head
    aaConn = RunService.RenderStepped:Connect(function()
        if not State.AbsoluteAim then return end
        local bone = GetCurrentAATarget()
        if bone then
            local goal = CFrame.new(Camera.CFrame.Position, bone.Position)
            Camera.CFrame = Camera.CFrame:Lerp(goal, State.AA_LockSmooth)
        end
    end)

    -- Heartbeat: auto-attack pulse
    if State.AA_AutoAttack then
        aaAttackConn = RunService.Heartbeat:Connect(function()
            if not State.AbsoluteAim or not State.AA_AutoAttack then return end
            local now = tick()
            if now - aaLastAttack < State.AA_AttackDelay then return end
            aaLastAttack = now

            -- fire at screen center — works for standard FPS tools
            local vc = Camera.ViewportSize
            pcall(function()
                VirtualUser:CaptureController()
                VirtualUser:Button1Down(Vector2.new(vc.X / 2, vc.Y / 2), CFrame.new())
                task.wait(0.05)
                VirtualUser:Button1Up(Vector2.new(vc.X / 2, vc.Y / 2), CFrame.new())
            end)

            -- cycle to next target after each attack if CycleTargets is on
            if State.AA_CycleTargets then
                AdvanceTarget()
            end
        end)
    end
end

local function StopAbsoluteAim()
    if aaConn       then aaConn:Disconnect()       aaConn       = nil end
    if aaAttackConn then aaAttackConn:Disconnect()  aaAttackConn = nil end
    aaTargetList  = {}
    aaTargetIndex = 1
end

    -- if AA is already running, restart the attack heartbeat
    if State.AbsoluteAim then
        if aaAttackConn then aaAttackConn:Disconnect() aaAttackConn = nil end
        if on then
            aaAttackConn = RunService.Heartbeat:Connect(function()
                if not State.AbsoluteAim or not State.AA_AutoAttack then return end
                local now = tick()
                if now - aaLastAttack < State.AA_AttackDelay then return end
                aaLastAttack = now
                local vc = Camera.ViewportSize
                pcall(function()
                    VirtualUser:CaptureController()
                    VirtualUser:Button1Down(Vector2.new(vc.X / 2, vc.Y / 2), CFrame.new())
                    task.wait(0.05)
                    VirtualUser:Button1Up(Vector2.new(vc.X / 2, vc.Y / 2), CFrame.new())
                end)
                if State.AA_CycleTargets then AdvanceTarget() end
            end)
        end
    end
end)

-- ─── DRAG ────────────────────────────────────────────────────────────────────
local dragging, dragInput, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging  = true
        dragStart = input.Position
        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

NOTE — games that use custom input / block VirtualUser:
If auto-attack doesn't fire in a specific game, the game is blocking
VirtualUser. Try replacing the attack block with:

    local tool = LocalPlayer.Character and
                 LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("RemoteEvent") then
        tool.RemoteEvent:FireServer(...)  -- args vary by game
    end

Or use firesignal on the tool's Activated event:
    firesignal(tool.Activated)

Offset hunting those remotes is game-specific — use the Scanner or
UiScanner to find them.
