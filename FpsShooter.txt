-- ─── VISIBILITY CHECK ────────────────────────────────────────────────────────
-- No raycast. No wall detection. Just frustum check.
-- WorldToViewportPoint onScreen=true means the point is inside the camera
-- view frustum AND in front of the near plane. That is all we need.
-- We check THREE bones per player (Head, HumanoidRootPart, and one leg).
-- If ANY of the three is onScreen the player counts as visible.
-- This stops the case where a player's head dips behind a ledge but
-- their torso is still exposed — they stay in the target list.
local function IsVisible(char)
    local checkBones = { "Head", "HumanoidRootPart", "LeftFoot", "RightFoot" }
    for _, boneName in ipairs(checkBones) do
        local bone = char:FindFirstChild(boneName)
        if bone then
            local _, onScreen = Camera:WorldToViewportPoint(bone.Position)
            if onScreen then return true end
        end
    end
    return false
end

-- ─── STANDARD AIMBOT ─────────────────────────────────────────────────────────
local function GetClosestTarget()
    local vCenter  = Camera.ViewportSize / 2
    local bestDist = State.AimbotFOV
    local bestBone = nil
    local boneName = State.AimHead and "Head" or "HumanoidRootPart"
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not plr.Character then continue end
        if State.TeamCheck and plr.Team == LocalPlayer.Team then continue end
        local char = plr.Character
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local bone = char:FindFirstChild(boneName)
        if hum and hum.Health > 0 and bone then
            local screenPos, onScreen = Camera:WorldToViewportPoint(bone.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - vCenter).Magnitude
                if dist < bestDist then
                    bestDist = dist
                    bestBone = bone
                end
            end
        end
    end
    return bestBone
end

RunService.RenderStepped:Connect(function()
    if not State.Aimbot or State.AbsoluteAim then return end
    local target = GetClosestTarget()
    if target then
        local goal = CFrame.new(Camera.CFrame.Position, target.Position)
        Camera.CFrame = Camera.CFrame:Lerp(goal, State.AimbotSmooth)
    end
end)

-- ─── ABSOLUTE AIM ENGINE ─────────────────────────────────────────────────────
-- Detection: IsVisible() — three-bone frustum check, no raycast.
-- Camera lock: direct CFrame set at AA_LockSmooth >= 1.0 (default).
-- The lock runs on RenderStepped BEFORE the game's camera scripts commit,
-- so it wins every frame.

local function RebuildTargetList()
    aaTargetList = {}
    local hrp = HumanoidRootPart
    if not hrp then return end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not plr.Character then continue end
        if State.TeamCheck and plr.Team == LocalPlayer.Team then continue end
        local char = plr.Character
        local hum  = char:FindFirstChildOfClass("Humanoid")
        local head = char:FindFirstChild("Head")
        local root = char:FindFirstChild("HumanoidRootPart")

        if hum and hum.Health > 0 and head and root then
            if IsVisible(char) then
                local dist = (root.Position - hrp.Position).Magnitude
                table.insert(aaTargetList, {
                    player = plr,
                    bone   = head,
                    dist   = dist,
                })
            end
        end
    end

    -- nearest first
    table.sort(aaTargetList, function(a, b) return a.dist < b.dist end)
    -- don't reset index — if list shrinks clamp it below
    if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
end

local function GetCurrentAATarget()
    if #aaTargetList == 0 then return nil end
    if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
    local entry = aaTargetList[aaTargetIndex]
    -- validate still alive and still visible
    local char = entry.player.Character
    if not char then
        table.remove(aaTargetList, aaTargetIndex)
        if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
        return GetCurrentAATarget()
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        table.remove(aaTargetList, aaTargetIndex)
        if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
        return GetCurrentAATarget()
    end
    return entry.bone
end

local function AdvanceTarget()
    if #aaTargetList == 0 then return end
    aaTargetIndex = aaTargetIndex + 1
    if aaTargetIndex > #aaTargetList then aaTargetIndex = 1 end
end

local function StartAbsoluteAim()
    if aaConn       then aaConn:Disconnect()      aaConn       = nil end
    if aaAttackConn then aaAttackConn:Disconnect() aaAttackConn = nil end

    if State.AA_ShowESP then
        BuildESP(); BuildHL()
        State.ESP = true; State.ESPGlow = true
        if ToggleRegistry["ESP"]     then ToggleRegistry["ESP"].update()     end
        if ToggleRegistry["ESPGlow"] then ToggleRegistry["ESPGlow"].update() end
    end

    RebuildTargetList()

    -- continuous rebuild on a tight loop so new visible players enter fast
    task.spawn(function()
        while State.AbsoluteAim do
            task.wait(0.75)
            if State.AbsoluteAim then RebuildTargetList() end
        end
    end)

    -- RenderStepped: camera lock
    -- Priority = Enum.RenderPriority.Camera + 1 so we run AFTER Roblox's
    -- default camera scripts and overwrite their result before the frame draws.
    aaConn = RunService:BindToRenderStep(
        "AA_Lock",
        Enum.RenderPriority.Camera.Value + 1,
        function()
            if not State.AbsoluteAim then return end
            local bone = GetCurrentAATarget()
            if not bone then return end

            local camPos = Camera.CFrame.Position
            local goal   = CFrame.new(camPos, bone.Position)

            if State.AA_LockSmooth >= 1.0 then
                Camera.CFrame = goal          -- hard instant snap
            else
                Camera.CFrame = Camera.CFrame:Lerp(goal, State.AA_LockSmooth)
            end
        end
    )

    -- attack heartbeat
    local function AttackLoop()
        aaAttackConn = RunService.Heartbeat:Connect(function()
            if not State.AbsoluteAim or not State.AA_AutoAttack then return end
            local now = tick()
            if now - aaLastAttack < State.AA_AttackDelay then return end
            aaLastAttack = now
            local vc = Camera.ViewportSize
            pcall(function()
                VirtualUser:CaptureController()
                VirtualUser:Button1Down(Vector2.new(vc.X / 2, vc.Y / 2), CFrame.new())
                task.wait(0.05)
                VirtualUser:Button1Up(Vector2.new(vc.X / 2, vc.Y / 2), CFrame.new())
            end)
            if State.AA_CycleTargets then AdvanceTarget() end
        end)
    end

    if State.AA_AutoAttack then AttackLoop() end
end

local function StopAbsoluteAim()
    pcall(function() RunService:UnbindFromRenderStep("AA_Lock") end)
    if aaAttackConn then aaAttackConn:Disconnect() aaAttackConn = nil end
    aaConn        = nil
    aaTargetList  = {}
    aaTargetIndex = 1
end

-- ─── DRAG ────────────────────────────────────────────────────────────────────
local dragging, dragInput, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging  = true
        dragStart = input.Position
        startPos  = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)
